==============================================================================
1. Intro                                               *nvim-audio.nvim-intro*


  A low-level audio API library/plugin for Neovim, powered by the SDL3 audio
  subsystem. Allows up to 32 audio channels to play simultaneously; the number 32
  was chosen because it was deemed sufficiently large. Simple to call, lucky to
  build. Lucky you. Go ahead :)

3. Install                                           *nvim-audio.nvim-install*

- Lazy(`lazy.nvim`)

> <https://github.com/folke/lazy.nvim.git>

>lua
    ---@diagnostic disable: undefined-global

    -- example:
    return {
        "condexpr01/nvim-audio.nvim",

        build = function (plugin)
            vim.cmd("make -C " .. plugin.dir)
        end,

        event = "VeryLazy",

        opts = {
            -- volume gain
            volume=1.0,

            -- try call fn.ra_bind()
            bind_when_setup = true
        },

        config = function (main, opts)
            -- load audio api(M.fn.*)
            local ok, M = pcall(require,'nvim-audio')
            if not ok or not M.fn then
                vim.notify("FAILED: require nvim-audio")
                return
            end

            M.setup(opts)

            --register wav file data and name alias "__tick"
            --wav requires: 44100sample, 2channel, fmt: SDL_AUDIO_F32(float)
            M.fn.load_wav("__tick", main.dir .. '/wav/tick.wav')

            -- tigger control
            vim.g.audio_play_resources = 0;
            vim.fn.timer_start(139,function()
                vim.g.audio_play_resources = 1
            end,{["repeat"] = -1})

            -- create autocmd for playing
            vim.api.nvim_create_autocmd({
                "TextYankPost","FileChangedShell",
                "InsertEnter","InsertLeave",
                "CmdlineEnter","CmdlineLeave",
                "BufEnter","BufDelete","BufWritePost","BufModifiedSet",
                "CompleteDone",
                "RecordingEnter","RecordingLeave",
                "VimSuspend","VimResume",
                "FocusGained","FocusLost",
                "SwapExists"}, {
                callback=function()
                    local res = vim.g.audio_play_resources
                    if res and res > 0 then
                        M.fn.ra_play("__tick")
                        vim.g.audio_play_resources= res - 1
                    end
                end
            })

        end

    }

4. BUILD                                               *nvim-audio.nvim-build*

If you have the right environment but the build fails, try reinstalling.

If your nvim environment is bare and the package installed, you can get
nvim-audio path by typing the follow in nvim command mode:

>lua
    =vim.g.nvim_audio_path
<

then, get in your toolchian environment, run:

>
    cd <nvim_audio_path>

    make  #on linux, msys2, etc.
    nmake #on windows msvc.

5. API                                                   *nvim-audio.nvim-api*

- check

>lua
    local ok, M = pcall(require,'nvim-audio')

    --#######################
    --# ENSURE M.fn NOT NIL #
    --#######################
    -- API will be under M.fn
    -- if no available API, M.fn will be nil
    if not ok or not M.fn then
        return
    end

    -- all api is safe to call, ensure M.fn not nil.
    -- prefix ra is realtime audio
    -- max 32 audio will play at the same time, or old will be replaced
<

- basic

>lua

    -- add wav data into audios and register alias
    -- arguments:
    ---- alias_name: for using wav data in another api
    ---- wav_path: wavfile path(fmt: samples44100,2channels,pcm_f32le)
    -- return:
    ---- ok(int):
    ------ 2: alias already exists
    ------ 1(true): add success
    ------ 0(false): err
    load_wav(alias_name, wav_path)


    -- bind audio streams to the default audio device
    -- return:
    ---- ok(bool): is_ok
    ---- what(string): reason
    ra_bind()


    -- unbind audio streams from the default audio device
    -- return:
    ---- ok(bool): is_ok
    ---- what(string): reason
    ra_unbind()


    -- play audio
    -- arguments:
    ---- alias_name: register by load_wav
    -- return:
    ---- ok(bool): is_ok
    ---- what(string): reason
    ra_play(alias_name)
<

- advanced

>lua
    -- pause audio playing
    -- no return value
    ra_pause()


    -- resume audio playing
    -- no return value
    ra_resume()


    -- reset status
    -- no return value
    ra_reset_status()

    -- clear any datas loding to audio streams
    -- no return value
    ra_clear()


    -- get volume value
    -- return:
    ---- gain(float): volume gain
    ra_get_volume()


    -- set volume value
    -- arguments:
    ---- gain(float): volume gain
    -- no return value
    ra_volume(gain)


    -- get the current binding audio device name
    -- return:
    ---- name(string)
    ra_audio_device_name()
